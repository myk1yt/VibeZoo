import os, json

appdata = os.environ.get('APPDATA', '')
base = os.path.join(appdata, 'Code', 'User', 'globalStorage')

found = []
if os.path.isdir(base):
    for item in os.listdir(base):
        full = os.path.join(base, item)
        # Check for mcp.json in subdirectories
        mcp_path = os.path.join(full, 'mcp.json')
        if os.path.isfile(mcp_path):
            found.append(mcp_path)
        # Also look for any JSON files directly
        if item.endswith('.json'):
            found.append(full)

results = []
if found:
    for p in found:
        sz = os.path.getsize(p)
        with open(p, 'r', encoding='utf-8') as f:
            content = f.read()
        results.append(f'FILE: {p} ({sz} bytes)\n{content[:800]}')
else:
    results.append('No global MCP configs found')

# Also check AppData\Roo-Code
extra = os.path.join(appdata, 'Roo-Code')
if os.path.isdir(extra):
    results.append(f'\nRoo-Code dir exists: {extra}')
    for e in os.listdir(extra):
        results.append(f'  {e}')
        if 'mcp' in e.lower() and e.endswith('.json'):
            p = os.path.join(extra, e)
            with open(p, 'r', encoding='utf-8') as f:
                results.append(f'  Content: {f.read()[:500]}')

output = '\n'.join(results)
with open('global_mcp_found.txt', 'w', encoding='utf-8') as f:
    f.write(output)
print('done')
